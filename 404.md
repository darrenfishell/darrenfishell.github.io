<script>
(function() {
  function initImageSwap() {
    const imageContainer = document.getElementById('inferno-image');
    const originalImage = 'https://upload.wikimedia.org/wikipedia/commons/thumb/3/3e/Gustave_Dor%C3%A9_-_Dante_Alighieri_-_Inferno_-_Plate_8_%28Canto_III_-_Abandon_all_hope_ye_who_enter_here%29.jpg/2820px-Gustave_Dor%C3%A9_-_Dante_Alighieri_-_Inferno_-_Plate_8_%28Canto_III_-_Abandon_all_hope_ye_who_enter_here%29.jpg';
    const luciferImage = 'https://upload.wikimedia.org/wikipedia/commons/a/ad/DVinfernoLuciferKingOfHell_m.jpg';
    let isSwapped = false;
    
    if (imageContainer) {
      imageContainer.style.cursor = 'pointer';
      imageContainer.addEventListener('click', function() {
        isSwapped = !isSwapped;
        const newImage = isSwapped ? luciferImage : originalImage;
        imageContainer.style.backgroundImage = `url('${newImage}')`;
      });
    }
  }
  
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initImageSwap);
  } else {
    initImageSwap();
  }
})();
</script>
